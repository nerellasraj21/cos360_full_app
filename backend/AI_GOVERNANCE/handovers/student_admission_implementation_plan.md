### Step 5: Modify generate_admission_number()

**File:** `app/service/student/admission_service.py` (lines 38-67)

```python
async def generate_admission_number(
    db: AsyncSession,
    admission_date: date,
    admission_type: str = "non_primary"
) -> str:
    """
    Generate admission number with type-specific prefixes:
    - Primary: P{YEAR}{SEQ} (e.g., P2024001)
    - Non-Primary: NP{YEAR}{SEQ} (e.g., NP2024001)
    """
    year = admission_date.year
    prefix = "P" if admission_type == "primary" else "NP"

    # Count admissions for this year AND type
    count_stmt = select(func.count(Admission.id)).where(
        and_(
            extract('year', Admission.admission_date) == year,
            Admission.admission_type == admission_type
        )
    )
    result = await db.execute(count_stmt)
    count = result.scalar() or 0

    sequence = count + 1
    admission_number = f"{prefix}{year}{sequence:03d}"

    # Check for collisions (existing collision detection code)
    check_stmt = select(Admission).where(Admission.admission_number == admission_number)
    existing = await db.execute(check_stmt)
    if existing.scalar_one_or_none():
        # Handle collision by incrementing
        sequence += 1
        admission_number = f"{prefix}{year}{sequence:03d}"

    return admission_number
```

from typing import List, Optional
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func
from sqlalchemy.orm import selectinload
from fastapi import HTTPException, status
from app.models.masters.caste_model import Caste
from app.models.masters.sub_caste_model import SubCaste
from app.schemas.masters.caste_schema import (
    CasteCreate, CasteUpdate, CasteDropdown,
    SubCasteCreate, SubCasteUpdate, SubCasteDropdown
)
from app.core.cache import cache_dropdown

# ========== Caste CRUD ==========

async def create_caste(db: AsyncSession, data: CasteCreate) -> Caste:
    """Create new caste"""
    # Check uniqueness
    stmt = select(Caste).where(func.lower(Caste.name) == func.lower(data.name))
    existing = await db.execute(stmt)
    if existing.scalar_one_or_none():
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Caste '{data.name}' already exists"
        )

    caste = Caste(**data.dict())
    db.add(caste)

    # Use flush()->select()->commit() pattern for multi-tenant
    await db.flush()
    result = await db.execute(select(Caste).where(Caste.id == caste.id))
    caste_out = result.scalar_one()
    await db.commit()

    return caste_out

async def get_all_castes(
    db: AsyncSession,
    active_only: bool = True,
    skip: int = 0,
    limit: int = 100
) -> List[Caste]:
    """Get all castes"""
    stmt = select(Caste)
    if active_only:
        stmt = stmt.where(Caste.is_active == True)
    stmt = stmt.offset(skip).limit(limit).order_by(Caste.name)

    result = await db.execute(stmt)
    return result.scalars().all()

@cache_dropdown(ttl=300)
async def get_castes_dropdown(
    db: AsyncSession,
    active_only: bool = True
) -> List[CasteDropdown]:
    """Get castes for dropdown"""
    castes = await get_all_castes(db, active_only, skip=0, limit=1000)
    return [CasteDropdown.from_orm(c) for c in castes]

async def get_caste_by_id(db: AsyncSession, caste_id: str) -> Caste:
    """Get caste by ID"""
    stmt = select(Caste).where(Caste.id == caste_id)
    result = await db.execute(stmt)
    caste = result.scalar_one_or_none()

    if not caste:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Caste not found"
        )

    return caste

async def update_caste(
    db: AsyncSession,
    caste_id: str,
    data: CasteUpdate
) -> Caste:
    """Update caste"""
    caste = await get_caste_by_id(db, caste_id)

    # Check name uniqueness if changed
    if data.name and data.name != caste.name:
        stmt = select(Caste).where(func.lower(Caste.name) == func.lower(data.name))
        existing = await db.execute(stmt)
        if existing.scalar_one_or_none():
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Caste '{data.name}' already exists"
            )

    for field, value in data.dict(exclude_unset=True).items():
        setattr(caste, field, value)

    # Use flush()->select()->commit() pattern
    await db.flush()
    result = await db.execute(select(Caste).where(Caste.id == caste_id))
    caste_out = result.scalar_one()
    await db.commit()

    return caste_out

## Critical Patterns to Follow

### 1. Multi-Tenant Migration Pattern

```bash
# Always in this order:
1. Create migration:
   alembic revision -m "description"

2. Apply to main schema:
   python test_scripts/migrate_tenants.py --schema cos360_master --action upgrade --target head

3. Sync all tenants:
   python test_scripts/migrate_tenants.py --action sync-all

4. Verify sync:
   python test_scripts/migrate_tenants.py --action diagnose
```

### 2. Database Refresh Pattern (CRITICAL)

```python
# CORRECT - flush() -> select() -> commit()
await db.flush()  # Write to DB
result = await db.execute(select(Model).where(Model.id == obj.id))
obj_out = result.scalar_one()
await db.commit()
return obj_out

# WRONG - commit() -> refresh()
await db.commit()
await db.refresh(obj)  # Schema context lost in multi-tenant!
```

**Why:** Multi-tenant schema context is lost after commit. Must use select() to retrieve with correct schema.

### 3. Enum Pattern

```python
# Model
import enum
from sqlalchemy import Enum

class MyEnum(enum.Enum):
    value1 = "value1"
    value2 = "value2"

class MyModel(BaseOrg):
    field = Column(Enum(MyEnum, name='myenum', create_type=False), nullable=True)

# Schema
from typing import Literal

field: Optional[Literal["value1", "value2"]] = None
```

### 4. Service CRUD Pattern

- **Uniqueness checks** before create
- **Dependency checks** before delete
- **Cache invalidation** after CUD operations
- Use `@cache_dropdown(ttl=300)` for dropdown methods
- Proper error handling with HTTPException
- Use flush()->select()->commit() pattern

### 5. Endpoint Pattern

- **Permission check:** `check_role_plan_permission_with_error()`
- **Rate limiting:** `@rate_limit_create("30 per minute")`, `@rate_limit_dropdown("100 per minute")`
- **Standard HTTP status codes:** 200, 201, 204, 400, 404
- **Consistent response models:** Read schemas for responses
